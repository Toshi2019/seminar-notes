\section{前提知識}

\begin{DFN}
    \(\cat\)を圏とする．\(\cat\)の
    \emph{生成対象系} (system of generators) とは，
    \(\cat\)の対象の族\(\left(G_i\right)_{i\in I}\)で
    \(I\)が小さな集合であり，射\(f\colon X\to Y\)に対し，
    \(\Hom_{\cat}(G_i,X)\to\Hom_{\cat}(G_i,Y)\)が
    どの\(i\in I\)についても同型となるならば
    \(f\)も同型となることをいう．
\end{DFN}


\begin{DFN}
    \emph{グロタンディーク圏} (Grothendieck category) とは，
    アーベル圏であって，小さな極限と小さな余極限，
    生成対象で小さなフィルター余極限が完全となるものをもつものをいう．
\end{DFN}

\section{ブラウンの表現定理}
\begin{DFN}
    \(\cD\)を三角圏で小さな直和を持つものとする．
    \(\cD\)の\emph{\(t\)生成系} (system of 
    \(t\)-generators) \(\cF\)とは，
    \(\cD\)の対象の小さな族で以下の条件を満たすものをいう．
    \begin{enumerate}[(i)]
        \item \(X\in\cD\)が，
        すべての\(C\in\cF\)に対し\(\Hom_{\cD}(C,X)\cong 0\)となるならば，\(X\cong 0\)である．
        \item 任意の\emph{可算}集合\(I\)と任意\(\cD\)の射の
        族\(\left(u_i\colon X_i\to Y_i\right)_{i\in I}\)に対し，
        写像\(
            \Hom_{\cD}(C,X_i)
            \overset{u_i}{\longrightarrow}
            \Hom_{\cD}(C,Y_i)
        \)がどの\(i\in I\)とどの\(C\in\cF\)に対しても\(0\)になるならば，
        写像\(
            \Hom_{\cD}(C,\oplus_i X_i)
            \overset{\oplus_i u_i}{\longrightarrow}
            \Hom_{\cD}(C,\oplus_i Y_i)
        \)がすべての\(C\in\cF\)に対し\(0\)になる．
        %\(\xymatrix{
        %        \Hom_{\cD}(C,\oplus_i X_i)
        %        \ar[r]^{\oplus_i u_i}&
        %        \Hom_{\cD}(C,\oplus_i Y_i)
        %    }\)
    \end{enumerate}
\end{DFN}

以下でブラウンの表現定理と呼ぶものは，実際にはその系である．
\cite[Cor. 10.5.3]{KS06}も参照．

\begin{THM}[ブラウンの表現定理]
    \(\cD\)を三角圏で小さな直和と\(t\)生成系を持つものとする．
    \(F\colon\cD\to\cD'\)を三角圏の間の三角関手とし，
    \(F\)は小さな直和と交換すると仮定する．
    このとき\(F\)の右随伴三角関手\(G\)が存在する．
\end{THM}

\begin{THM}[{\cite[Th. 14.3.1]{KS06}参照}]
    \(\cC\)をグロタンディークアーベル圏とする．
    \begin{enumerate}[(a)]
        \item 圏\(\Domp(\cC)\)は小さな直和と\(t\)生成系をもつ．
        \item \(\cD\)を三角圏，\(F\colon\Komp(\cC)\to\cD\)を
        三角圏の間の関手とする．
        このとき\(F\)は右局所化\(
            \Rdf{F}\colon \Domp(\cC)\to\cD
        \)をもつ．
    \end{enumerate}
\end{THM}

\begin{THM}\label{757}
    \(\cC\)と\(\cC'\)をグロタンディーク圏とし，
    \(\rho\colon\cC\to\cC'\)を左完全関手とする．
    次を仮定する．
    \begin{enumerate}[(i)]
        \item \(\rho\)のコホモロジー次元は有限である．\label{757-11}
        \item \(\rho\)は小さな直和と交換する．\label{757-12}
        \item \(\cC\)の入射的対象の小さな直和は\(\rho\)に対し非輪状である．\label{757-13}
    \end{enumerate}
    このとき
    \begin{enumerate}[(a)]
        \item 関手\(\Rdf{\rho}\colon\Domp(\cC)\to\Domp(\cC')\)は小さな直和と交換する．\label{757-21}
        \item 関手\(\Rdf{\rho}\colon\Domp(\cC)\to\Domp(\cC')\)は右随伴\(\rho^{!}\colon\Domp(\cC')\to\Domp(\cC)\)をもつ．\label{757-22}
        \item 関手\(\rho^{!}\)は関手\(\rho^{!}\colon\Dompl(\cC')\to\Dompl(\cC)\)をひきおこす．\label{757-23}
    \end{enumerate}
\end{THM}


\section{ヴェルディエ双対}

\subsection{右随伴の存在}

\(f\colon X\to Y\)を局所コンパクト空間の射とし，\(\kk\)を体とする．
\(
    \Rdf{f_{!}}\colon \Mod(\kk_X)\to\Mod(\kk_Y)
\)の右随伴\(f^{!}\)を
例外逆像という．
これの存在を証明するのにブラウンの表現定理を応用する．

\begin{HPT}\label{1141}
    局所コンパクト空間とその間の射はすべて，
    c柔軟次元が有限であるとする．
\end{HPT}

\begin{PRP}\label{1115}
    \begin{enumerate}[(i)]
        \item \(f\colon X\to Y\)を局所コンパクト空間の射とし，
        \(F\in\Mod(\kk_X)\)とする．\(Y\)の開集合\(V\)に対し\[
            \Gamma(V;f_{!}F)
            \cong
            \indlim[Z]\Gamma_{Z}(f^{-1}(V);F)
        \]が成り立つ．
        ただし\(Z\)は\(f^{-1}(V)\)の閉集合で\(
            \mres[f]{Z}\colon Z\to V
        \)が固有射となるものの族を走る．
        とくに\(
            \Gamma_{c}(X;F)
            \cong
            \indlim[K]\Gamma_{K}(X;F)
        \)が成り立つ．\label{1115-1}
        ただし\(K\)は\(X\)のコンパクト部分集合の族を走る．
        \item \(f\)が\(\supp(F)\)上固有ならば，
        \(f_{!}F\simarr f_{\ast}F\)が成り立つ．
        特に\(f\)が固有ならば\(f_{!}\simarr f_{\ast}\)である．\label{1115-2}
        \item 関手\(f_{!}\)は左完全であり，
        小さなフィルター帰納極限と交換する．\label{1115-3}
        \item \(g\colon Y\to Z\)を局所コンパクト空間の間の
        連続写像とする．
        このとき\(g_{!}f_{!}\cong (g\circ f)_{!}\)である．\label{1115-4}
        %\item \(i_{U}\colon U\hookrightarrow X\)を開埋め込みとする．このとき\label{1115-5}
    \end{enumerate}
\end{PRP}
\begin{PRP}\label{1124}
    c柔軟層の小さなフィルター帰納極限はc柔軟である．
    特に，c柔軟層の小さな直和はc柔軟である．
\end{PRP}

\begin{THM}
    \(f\colon X\to Y\)を局所コンパクト空間の間の連続写像とする．
    このとき，関手\(
        \Rdf{f_{!}}\colon
        \Dompl(\kk_{X})\to\Dompl(\kk_{Y})
    \)は右随伴をもつ．
\end{THM}
\begin{proof}
    \(\Mod(\kk_X)\)はグロタンディーク圏である．
    仮定\ref{1141}より定理\ref{757}の仮定\eqref{757-11}が成り立つ．
    命題\ref{1115} \eqref{1115-3}より
    \(f_!\)は左完全であり定理\ref{757}の仮定\eqref{757-12}をみたす．
    命題\ref{1124}より定理\ref{757}の仮定\eqref{757-13}が成り立つ．
    したがって定理\ref{757}より\(
        \Rdf{f_{!}}\colon \Dompl(\kk_X)\to\Dompl(\kk_Y)
    \)の右随伴関手\(
        f^{!}\colon\Dompl(\kk_Y)\to\Dompl(\kk_X)
    \)が存在する．
\end{proof}

\subsection{ヴェルディエ双対の性質}

写像\(f\colon X\to Y\)があるとき，
グラフを用いてその写像を分解することができる．
\[
    f\colon X\hookrightarrow X\times Y\twoheadrightarrow Y.
\]
ヴェルディエ双対の自然性から，
分解した後の閉埋め込みと射影に関して計算ができればよい．

\begin{PRP}
    \(f\colon X\to Y\)を\(X\)から\(Y\)への閉埋め込みとし，
    \(X\)と同型になる\(Y\)の閉部分集合を\(Z\)とする．
    このとき\[
        f^{!}(\blk)\cong f^{-1}\RG_{Z}(\blk).
    \]
\end{PRP}

\begin{proof}
    \(F\in\Dompb(\kk_X)\), \(G\in\Dompl(\kk_Y)\)とする．
    \begin{align*}
        \Hom_{\kk_X}(F,f^{!}G)
        &\cong \Hom_{\kk_Y}(\Rdf{f_!}F,G)\\
        &\cong \Hom_{\kk_Y}(\Rdf{f_!}F\tens[]\kk_{Z},G)\\
        &\cong \Hom_{\kk_Y}(\Rdf{f_!}F,\RG_{Z}G)\\
        &\cong \Hom_{\kk_Y}(f^{-1}\Rdf{f_!}F,f^{-1}\RG_{Z}G)\\
        &\cong \Hom_{\kk_Y}(F,f^{-1}\RG_{Z}G)
    \end{align*}
    が\(F\in\Dompb(\kk_X)\)に関して自然に成り立つ．
    よって米田の補題から結論が従う．
\end{proof}
